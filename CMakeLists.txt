cmake_minimum_required(VERSION 3.16)
project(embedded_ci LANGUAGES C)
include(FetchContent)
FetchContent_Declare(unity GIT_REPOSITORY https://github.com/ThrowTheSwitch/Unity.git GIT_TAG v2.6.1)
FetchContent_MakeAvailable(unity)
add_library(firmware firmware.c)
add_executable(firmware_elf main.c mock.c)
set_target_properties(firmware_elf PROPERTIES OUTPUT_NAME firmware SUFFIX .elf)
target_link_libraries(firmware_elf PRIVATE firmware)
target_link_options(firmware_elf PRIVATE -Wl,-Map=firmware.map)
add_executable(firmware_test test.c mock.c)
target_link_libraries(firmware_test PRIVATE firmware unity)
enable_testing()
add_test(NAME firmware COMMAND firmware_test)
